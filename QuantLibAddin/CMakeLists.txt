# QuantLibAddin/CMakeLists.txt
# Builds the QuantLibObjects static library (the QuantLib/ObjectHandler wrapper).
#
# This directory can be configured in TWO ways:
#
#   * As part of the top-level build (add_subdirectory from the parent), where
#     ql_library and ObjectHandler::ohlib are in-tree targets created by the
#     QuantLib and ObjectHandler subdirectories the root CMakeLists.txt adds.
#
#   * Standalone — `cmake -S QuantLibAddin -B <build>` — to build just the
#     QuantLibAddin libraries (QuantLibObjects, AddinCpp) and the QLADemo client
#     against *prebuilt* QuantLib and ObjectHandler.  The prerequisites are
#     located with find_package(... CONFIG); point CMAKE_PREFIX_PATH (or the
#     per-package QuantLib_DIR / ObjectHandler_DIR) at their build trees.  Their
#     generic build settings come from gensrc/cmake/commonSettings.cmake (gensrc
#     is QuantLibAddin's upstream dependency, so referring to it does not violate
#     the project hierarchy).
#
# Either way QuantLibObjects links the namespaced targets QuantLib::QuantLib and
# ObjectHandler::ohlib, which resolve to the in-tree aliases in the integrated
# build and to the imported targets in the standalone build.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.15)
    # Required for CMAKE_MSVC_RUNTIME_LIBRARY (CMP0091) — must precede project().
    cmake_policy(SET CMP0091 NEW)
    project(QuantLibAddin CXX)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../gensrc/cmake/commonSettings.cmake")

    # Select the Boost runtime variant to match the CRT *before* the prerequisite
    # package configs run, and find Boost first so that the QuantLib::QuantLib /
    # ObjectHandler::ohlib imported targets (which list Boost as a PUBLIC link
    # dependency) resolve cleanly when their *Targets.cmake files are loaded.
    if(MSVC)
        if(NOT MSVC_LINK_DYNAMIC_RUNTIME)
            set(Boost_USE_STATIC_RUNTIME ON)
        endif()
        set(Boost_USE_STATIC_LIBS ON)
    endif()
    find_package(Boost 1.58.0 REQUIRED COMPONENTS filesystem serialization)

    # Consume the prebuilt prerequisites from their build trees.  Both must have
    # been built with the SAME toolset + CRT (static/dynamic) + configuration as
    # this build, or the static libraries will not link (see the LNK2038 note in
    # md/build_cmake.md).
    find_package(QuantLib CONFIG REQUIRED)
    find_package(ObjectHandler CONFIG REQUIRED)
endif()

# -----------------------------------------------------------------------
# gensrc — optional code-generation step (the "Full" build), standalone only.
#
# Many QuantLibObjects sources (qlo/valueobjects/*, qlo/serialization/**,
# qlo/enumerations/register/*) and the AddinCpp sources (Addins/Cpp/*) are
# autogenerated by gensrc and are not committed to the repository, so on a
# pristine clone they do not exist yet.  CMake checks that the sources listed
# in add_library() exist when the target is *created*, i.e. at configure time;
# a build-time-only target would run too late and the first configure of
# QuantLibObjects would fail with "Cannot find source file".  When RUN_GENSRC is
# ON we therefore run gensrc here, DURING configuration, so the generated
# sources exist before the targets below are created.  (A build-time qlgensrc
# target is also defined further down, so an incremental build refreshes the
# generated sources before compiling.)
#
# This block is guarded by the standalone check, so in the integrated (root)
# build it is skipped — the root CMakeLists.txt owns the gensrc wiring.  gensrc
# is QuantLibAddin's upstream dependency, and --oh_dir points at the upstream
# ObjectHandler tree, so this references no downstream project.  The Excel
# target (x) is intentionally omitted (we generate -pvels, not -a): the Excel
# code writes into the downstream QuantLibXL tree, which a QuantLibAddin build
# must not touch — QuantLibXL runs its own gensrc.
# -----------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    option(RUN_GENSRC "Run gensrc to regenerate addin source before building" OFF)

    if(RUN_GENSRC)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)

        # The QuantLibAddin gensrc command (C++ addin, ValueObjects,
        # Enumerations, Loops, Serialization), run from QuantLibAddin/gensrc with
        # --oh_dir pointing back at the upstream ObjectHandler tree.  The
        # generator script itself lives in the top-level gensrc directory.
        set(_qlgensrc_cmd ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/../gensrc/gensrc.py" -pvels --oh_dir=../../ObjectHandler)
        set(_qlgensrc_wd  "${CMAKE_CURRENT_SOURCE_DIR}/gensrc")

        # Configure-time generation, recorded with a stamp file so it only runs
        # once per build tree.  Delete the stamp (or configure into a clean build
        # directory) to force regeneration.
        set(_qlgensrc_stamp "${CMAKE_CURRENT_BINARY_DIR}/qlgensrc.configured.stamp")
        if(NOT EXISTS "${_qlgensrc_stamp}")
            message(STATUS "RUN_GENSRC: running gensrc for QuantLibAddin (configure-time)")
            execute_process(
                COMMAND ${_qlgensrc_cmd}
                WORKING_DIRECTORY ${_qlgensrc_wd}
                RESULT_VARIABLE _qlgensrc_rc
                OUTPUT_VARIABLE _qlgensrc_out
                ERROR_VARIABLE  _qlgensrc_out)
            if(NOT _qlgensrc_rc EQUAL 0)
                message(FATAL_ERROR "gensrc (QuantLibAddin) failed with code ${_qlgensrc_rc}:\n${_qlgensrc_out}")
            endif()
            file(WRITE "${_qlgensrc_stamp}" "gensrc ran at configure time\n")
        endif()
    endif()
endif()

# Ensure Boost components are available.
if(MSVC)
    if(NOT MSVC_LINK_DYNAMIC_RUNTIME)
        set(Boost_USE_STATIC_RUNTIME ON)
    endif()
    set(Boost_USE_STATIC_LIBS ON)
endif()
find_package(Boost 1.58.0 REQUIRED COMPONENTS filesystem serialization)

target_include_directories(QuantLibObjects PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../ObjectHandler>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../QuantLib>
)

target_compile_definitions(QuantLibObjects PRIVATE
    _LIB
)

# Suppress the qlo/auto_link.hpp #pragma comment(lib, ...) directive; cmake
# manages linking.  PUBLIC so every consumer of QuantLibObjects (e.g. the XLL)
# inherits it too.
target_compile_definitions(QuantLibObjects PUBLIC QLADDIN_NO_AUTO_LINK)

# Link the namespaced prerequisite targets.  These resolve to the in-tree
# ALIAS targets in the integrated (root) build and to the imported targets from
# find_package(QuantLib/ObjectHandler CONFIG) in the standalone build.  Linking
# them also propagates their interface include directories — in particular the
# QuantLib *build* tree that holds the generated ql headers — so QuantLibObjects
# compiles in both modes.
target_link_libraries(QuantLibObjects PUBLIC
    QuantLib::QuantLib
    ObjectHandler::ohlib
    Boost::filesystem
    Boost::serialization
)

# Namespaced alias so consumers (e.g. QuantLibXL) can link the same
# QuantLibAddin::QuantLibObjects name whether QuantLibAddin is added in-tree or
# imported via find_package(QuantLibAddin CONFIG) in a standalone build.
add_library(QuantLibAddin::QuantLibObjects ALIAS QuantLibObjects)

# -----------------------------------------------------------------------
# gensrc — build-time regeneration target (the "Full" build), standalone only.
#
# Companion to the configure-time generation near the top of this file: the
# qlgensrc custom target re-runs gensrc on an incremental build so that edits to
# the gensrc metadata/templates refresh the generated qlo / AddinCpp sources
# before compilation.  QuantLibObjects and AddinCpp are wired to depend on it
# (AddinCpp's dependency is added after its subdirectory below), mirroring the
# qlgensrc dependency in the integrated build.  Skipped in the integrated build,
# where the root CMakeLists.txt owns this wiring.
# -----------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND RUN_GENSRC)
    add_custom_target(qlgensrc
        COMMAND ${_qlgensrc_cmd}
        WORKING_DIRECTORY ${_qlgensrc_wd}
        COMMENT "Running gensrc for QuantLibAddin"
        VERBATIM)
    add_dependencies(QuantLibObjects qlgensrc)
endif()

# AddinCpp static library and QLADemo client executable

add_subdirectory(Addins/Cpp)

add_subdirectory(Clients/CppSwapOut)

# AddinCpp's sources are autogenerated too, so in a standalone Full build make
# it depend on qlgensrc as well (the target was created above).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND RUN_GENSRC)
    add_dependencies(AddinCpp qlgensrc)
endif()

# -----------------------------------------------------------------------
# Build-tree package export (standalone only).
#
# When QuantLibAddin is configured on its own, write a CMake package config into
# the build tree so a downstream *standalone* build (QuantLibXL) can consume the
# QuantLibObjects wrapper library we just built with
#     find_package(QuantLibAddin CONFIG)
# pointed at this build directory — with no install step.  The exported imported
# target is QuantLibAddin::QuantLibObjects.
#
# Skipped in the integrated (root) build, where QuantLibXL links the in-tree
# QuantLibAddin::QuantLibObjects alias directly.  QuantLibObjects links the
# upstream QuantLib::QuantLib and ObjectHandler::ohlib targets, so the generated
# config re-finds those upstream packages (see QuantLibAddinConfig.cmake.in);
# this block references no downstream project and respects the dependency
# hierarchy.
# -----------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    export(TARGETS QuantLibObjects
        NAMESPACE QuantLibAddin::
        FILE "${CMAKE_CURRENT_BINARY_DIR}/cmake/QuantLibAddinTargets.cmake")

    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/cmake/QuantLibAddinConfig.cmake.in"
        "${CMAKE_CURRENT_BINARY_DIR}/cmake/QuantLibAddinConfig.cmake"
        COPYONLY)
endif()
